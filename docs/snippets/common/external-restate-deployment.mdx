```yaml expandable service-deployment.yaml
apiVersion: restate.dev/v1beta1
kind: RestateDeployment
metadata:
  name: service
spec:
  replicas: 1
  restate:
    register:
      cloud: my-cloud-environment
    tunnelMode: external
  selector:
    matchLabels:
      app: service
  template:
    metadata:
      labels:
        app: service
    spec:
      containers:
        - name: service
          image: <registry>/my-restate-service:0.0.1
          env:
            - name: PORT
              value: "9080"
          ports:
            - containerPort: 9080
              name: restate
```
